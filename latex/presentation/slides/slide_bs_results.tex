\section{Black-Scholes Experiment's Results}

\begin{frame}{Black-Scholes experimental setup}

Given zero interest rate and
zero dividend yield, the log-price of the asset follows a Brownian Motion process:

\begin{equation}
\begin{aligned}
dX_t^j
&= \mu_j dt -\frac{1}{2}\sigma_j^2\,dt
+ \sigma_j\,dW_t .
\end{aligned}
\end{equation}


\vspace{0.1cm}

\begin{itemize}
    \item $\mu_j = 0$: drift
    \item $X_0^j = 0$: initial log-price
    \item $\sigma_j \in (0.1,1.2)$: randomly sampled volatility
    \item $T=1$ year with $N=252$ daily observations
    \item $J=100{,}000$ independent trajectories
\end{itemize}

\vspace{0.15cm}

The process is discretized over $N$ uniform time steps:

\begin{equation}
\begin{aligned}
X_{t+dt}^j
&= X_t^j + -\frac{1}{2}\sigma_j^2dt
+\sigma_j\sqrt{dt}\,Z_t,
\qquad Z_t\sim\mathcal{N}(0,1).
\end{aligned}
\end{equation}

\end{frame}


\begin{frame}{Black--Scholes forecasting experiment}

\textbf{Objective}

Learn the conditional distribution of the log-price at
$T+\tau$ given the trajectory $j$ observed up to $T$.

\vspace{0.15cm}

\begin{equation}
\begin{aligned}
X_{T+\tau}^j \mid c_j
&\sim
\mathcal{N}
\left(
X_T^j-\frac{1}{2}\sigma_j^2\Delta t,
\,
\sigma_j\sqrt{\Delta t}
\right)
\end{aligned}
\end{equation}

where $\Delta t = dt \times \tau$.

\vspace{0.45cm}

\textbf{Experimental configuration}
\begin{itemize}
    \item \textbf{n-steps ahead}: $\tau = 10$
    \item \textbf{Condition:} $c_j=[X_0^j,X_{dt}^j,\ldots,X_T^j]$
    \item \textbf{Ground truth:} analytical and Monte Carlo Gaussian distribution
    \item \textbf{Models:} ForGAN with BCE, ForWGAN with Wasserstein loss and Scoring Rule ForGAN
\end{itemize}
\end{frame}

\begin{frame}{Distribution comparison and statistical testing}

Given two distributions $P$ and $Q$ over the same support:
\vspace{0.5cm}
\begin{itemize}

    \item \textbf{Total Variation} $\rightarrow TV(P,Q)=\frac{1}{2}\sum_i |p_i-q_i|$\\
    Measures the difference in probability
    assigned to the same event. 
    \vspace {0.4cm}
    \item \textbf{Hellinger distance} $\rightarrow H(P,Q)=\frac{1}{\sqrt{2}}\sqrt{\sum_i(\sqrt{p_i}-\sqrt{q_i})^2}$\\
    Bounded in $[0,1]$ it is more sensitive to low probability events.
    \vspace {0.4cm}
    \item \textbf{Jensen--Shannon} $\rightarrow JS(P,Q)=\frac{1}{2}KL(P\|M)+\frac{1}{2}KL(Q\|M),\qquad M=\frac{P+Q}{2}$\\
    Symmetric and bounded divergence based on KL divergence.
    \vspace{0.4cm}
    \item \textbf{Wasserstein distance} $\rightarrow W_1(P,Q)=\Delta_x\sum_i|CDF_P(i)-CDF_Q(i)|$\\
    \emph{geometry-aware}: it accounts for how far
    probability mass must be moved between bins to match.

    \vspace{0.4cm}
    \item \textbf{Kolmogorov--Smirnov (KS) test} $\rightarrow D_{p,q}=\sup_x|\hat F_p(x)-\hat G_q(x)|$\\
    Tests whether the two empirical distributions differ. Under $H_0:\;F(x)=G(x)\quad\forall x,$ 
    the asymptotic p-value is obtained from the Kolmogorov distribution.

\end{itemize}
\end{frame}


\begin{frame}{ForGAN vs ForWGAN vs SRForGAN: Results}

\textbf{Experiment:} same training/test data; only the architecture and
training objective are changed from ForGAN, ForWGAN and SRForGAN.


\begin{table}
\centering
\begin{tabular}{lccc}
\hline
\textbf{Metric} & \textbf{ForGAN} & \textbf{ForWGAN} & \textbf{SRForGAN} \\
\hline
KS Test no-rejection (\%) & 25.00 & 39.00 & 99.00 \\
EMD Distance            & 0.00855 & 0.00409 & 0.0018 \\
JS Divergence             & 0.4378 & 0.3602 & 0.1365 \\
Hellinger Distance      & 0.4001 & 0.3210 & 0.1178 \\
TV Distance             & 0.4129 & 0.3339 & 0.1099 \\
\hline
\end{tabular}
\end{table}


\begin{itemize}
    \item \textbf{ForWGAN improves all distributional metrics} with respect to ForGAN.
    
    \item KS test no-rejection increases from \textbf{25\%} to \textbf{39\%}.
    
    \item JS distance decreases from \textbf{0.4378} to \textbf{0.3602},
    indicating a closer match to the target distributions.
    
    \item Despite the improvement, the Wassertein model still struggles to accurately
    reproduce the target distributions, motivating introduction of scoring rule ForGAN.

    \item \textbf{SRForGAN achieves 99\% KS test no-rejection rate} and the lowest values for all other metrics, 
    avoiding the limitations of adversarial training like mode collapse.
\end{itemize}


\end{frame}
