\section{Black-Scholes Experiment's Results}

\begin{frame}{Black-Scholes experimental setup}

Under the risk-neutral measure with zero interest rate and
zero dividend yield, the log-price follows

\begin{equation}
\begin{aligned}
dX_t^j
&= -\frac{1}{2}\sigma_j^2\,dt
+ \sigma_j\,dW_t .
\end{aligned}
\end{equation}

\vspace{0.1cm}

The process is discretized over $N$ uniform time steps:

\begin{equation}
\begin{aligned}
X_{t+dt}^j
&= X_t^j
-\frac{1}{2}\sigma_j^2dt
+\sigma_j\sqrt{dt}\,Z_t,
\qquad Z_t\sim\mathcal{N}(0,1).
\end{aligned}
\end{equation}

\vspace{0.15cm}

\begin{itemize}
    \item $X_0^j = 0$: initial log-price
    \item $\sigma_j \in (0.1,1.2)$: randomly sampled volatility
    \item $T=1$ year with $N=252$ daily observations
    \item $J=100{,}000$ independent trajectories
\end{itemize}


\end{frame}


\begin{frame}{Black--Scholes forecasting experiment}

\textbf{Objective}

Learn the conditional distribution of the log-price at
$T+\tau$ given the trajectory $j$ observed up to $T$.

\vspace{0.15cm}

\begin{equation}
\begin{aligned}
X_{T+\tau}^j \mid c_j
&\sim
\mathcal{N}
\left(
X_T^j-\frac{1}{2}\sigma_j^2\Delta t,
\,
\sigma_j\sqrt{\Delta t}
\right)
\end{aligned}
\end{equation}

where $\Delta t = dt \times \tau$.

\vspace{0.15cm}

\textbf{Experimental configuration}
\begin{itemize}
    \item \textbf{Initial price}: $X_0 = 0.0$
    \item \textbf{Volatility range}: $\sigma \in (0.1, 1.2)$
    \item \textbf{Time horizon}: $T = 1.0 $ yearly horizon
    \item \textbf{Number of time steps}: $N = 252$
    \item \textbf{Number of simulated paths}: $J = 100{,}000$
    \item \textbf{n-steps ahead}: $\tau = 10$
    \item \textbf{Condition:} $c_j=[X_0^j,X_{dt}^j,\ldots,X_T^j]$
    \item \textbf{Ground truth:} analytical and Monte Carlo Gaussian distribution
    \item \textbf{Models:} ForGAN with BCE and WGAN-GP losses
\end{itemize}
\end{frame}

\begin{frame}{Statistical distance metrics}
Given two discrete probability distributions $P=(p_1,\ldots,p_B)$ and $Q=(q_1,\ldots,q_B)$ over the same support.

\begin{columns}[T,onlytextwidth]

%-----------------------------------------------------------
% LEFT COLUMN
%-----------------------------------------------------------
\begin{column}{0.48\textwidth}

\textbf{Total Variation (TV)}

\begin{equation*}
\delta(P,Q)
=
\frac{1}{2}\sum_i |p_i-q_i|
\end{equation*}

Measures the maximum difference in probability
assigned to the same event. 
%It is \textbf{bin-blind}: it does not account for the distance between bins.

\vspace{0.2cm}

\textbf{Hellinger distance}

\begin{equation*}
H(P,Q)
=
\frac{1}{\sqrt{2}}
\sqrt{\sum_i
\left(\sqrt{p_i}-\sqrt{q_i}\right)^2}
\end{equation*}

Bounded in $[0,1]$ it is more sensitive to differences in the
tails than TV distance.

\end{column}

%-----------------------------------------------------------
% RIGHT COLUMN
%-----------------------------------------------------------
\begin{column}{0.48\textwidth}

\textbf{Jensen--Shannon (JS)}

\begin{equation*}
JS(P,Q)
=
\frac{1}{2}KL(P\|M)
+
\frac{1}{2}KL(Q\|M),
\qquad
M=\frac{P+Q}{2}
\end{equation*}

Symmetric and smoothed version of KL divergence.

\vspace{0.2cm}

\textbf{Wasserstein distance}

\begin{equation*}
W_1(P,Q)
=
\sum_i
\left|
CDF_P(i)-CDF_Q(i)
\right|
\end{equation*}
\textbf{geometry-aware}: it accounts for how far
probability mass must be moved between bins to match.

\end{column}

\end{columns}

\end{frame}

\begin{frame}{Kolmogorov--Smirnov (KS) test}

The KS test is a \textbf{distribution-free} and \textbf{non-parametric} test that measures the maximum discrepancy between an
empirical CDF $F_n(x)$ and a reference CDF $F(x)$:


\begin{equation}
D
=
\max_k
\left|
CDF_p(i)
-
CDF_q(i)
\right|.
\end{equation}


Large $D$ indicates a larger discrepancy and provides
evidence against $H_0$.

\vspace{0.3cm}

Under $H_0: CDF_p(i)=CDF_q(i)$, the \textbf{p-value} is can be computed as:

\begin{equation}
P\!\left(\sqrt{n}D\geq t\right)
=
2\sum_{k=1}^{\infty}
(-1)^{k-1}e^{-2k^2t^2}.
\end{equation}

For the discretized distributions, the effective sample size is set to

\[
\boxed{n=B}
\]

where $B$ is the number of bins.

\end{frame}


\begin{frame}{ForGAN vs ForWGAN vs SRForGAN: Results}

\textbf{Experiment:} same training/test data; only the architecture and
training objective are changed from ForGAN, ForWGAN and SRForGAN.


\begin{table}
\centering
\begin{tabular}{lccc}
\hline
\textbf{Metric} & \textbf{ForGAN} & \textbf{ForWGAN} & \textbf{SRForGAN} \\
\hline
KS Test Acceptance (\%) & 25.00 & 39.00 & 100.00 \\
EMD Distance            & 0.00855 & 0.00409 & 0.0018 \\
JS Divergence             & 0.4378 & 0.3602 & 0.1365 \\
Hellinger Distance      & 0.4001 & 0.3210 & 0.1178 \\
TV Distance             & 0.4129 & 0.3339 & 0.1099 \\
\hline
\end{tabular}
\end{table}


\begin{itemize}
    \item \textbf{ForWGAN improves all distributional metrics} with respect to ForGAN.
    
    \item KS test acceptance increases from \textbf{25\%} to \textbf{39\%}.
    
    \item JS distance decreases from \textbf{0.4378} to \textbf{0.3602},
    indicating a closer match to the target distributions.
    
    \item Despite the improvement, the Wassertein model still struggles to accurately
    reproduce the target distributions, motivating introduction of scoring rule ForGAN.

    \item \textbf{SRForGAN achieves 100\% KS test acceptance} and the lowest values for all other metrics, 
    avoiding the limitations of adversarial training like mode collapse.
\end{itemize}


\end{frame}

\begin{frame}{Limitations of GANs}
\textbf{Practical limitations of adversarial training}

\vspace{0.5cm}

\begin{itemize}
    \item \textbf{Non-convergent optimization:}
    the GAN objective is a non-convex two-player game; stochastic
    gradient descent is not guaranteed to converge to the Nash equilibrium.

    \item \textbf{Mode collapse:}
    the generator may concentrate on a small subset of the target
    distribution, failing to reproduce the full probabilistic uncertainty.

    \item \textbf{Underdispersion in forecasting:}
    for a fixed historical condition, different noise draws can produce
    very similar outputs, resulting in a predictive distribution with
    artificially low variance.

    \item \textbf{Vanishing gradients:}
    when the discriminator becomes too effective, the generator may
    receive weak or uninformative gradients precisely when it needs
    corrective feedback.

    \item \textbf{Diversity is not explicitly rewarded:}
    the original adversarial objective provides no direct mechanism
    encouraging the generator to preserve the dispersion of the target
    distribution.
\end{itemize}


\end{frame}
